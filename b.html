import requests import pandas as pd from datetime import datetime import time SERVICE_URL = "https://gis.bnpb.go.id/arcgis/rest/services/BNPB_Web/Kejadian_Bencana/FeatureServer/0/query" def get_all_data(): all_features = [] offset = 0 while True: params = { "where": "1=1", "outFields": "*", "returnGeometry": "false", "resultOffset": offset, "resultRecordCount": 1000, "f": "json" } resp = requests.get(SERVICE_URL, params=params) data = resp.json() if "features" in data: all_features.extend(data["features"]) if not data.get("exceededTransferLimit"): break offset += 1000 return all_features while True: print(f"[{datetime.now()}] Mengambil data...") features = get_all_data() df = pd.DataFrame([f["attributes"] for f in features]) df.to_csv("bencana_live.csv", index=False) print(f"Tersimpan {len(df)} rekaman. Menunggu 5 menit...") time.sleep(300) # 5 menit
